SELECT
    -- volumes et période
    COUNT(*)                                                   AS nb_lignes,
    COUNT(DISTINCT num_automate)                               AS nb_gab,
    COUNT(DISTINCT date_operation)                             AS nb_jours,
    MIN(date_operation)  AS date_min,      MAX(date_operation)  AS date_max,
    MIN(heure_operation) AS heure_min,     MAX(heure_operation) AS heure_max,
    MIN(montant_ope)     AS montant_min,   MAX(montant_ope)     AS montant_max,
    MIN(encaisse_theorique) AS encaisse_min, MAX(encaisse_theorique) AS encaisse_max,

    -- types d'opération
    SUM(CASE WHEN lib_ope = 'Retrait'              THEN 1 ELSE 0 END) AS nb_retrait,
    SUM(CASE WHEN lib_ope = 'Autorisation-Retrait' THEN 1 ELSE 0 END) AS nb_autorisation_retrait,
    SUM(CASE WHEN lib_ope = 'Capture de Carte'     THEN 1 ELSE 0 END) AS nb_capture,
    SUM(CASE WHEN lib_ope NOT IN ('Retrait', 'Autorisation-Retrait', 'Capture de Carte') OR lib_ope IS NULL
             THEN 1 ELSE 0 END)                                       AS nb_autre_operation,

    -- inaboutis et porteurs
    SUM(CASE WHEN CAST(top_annul AS INT) = 1 THEN 1 ELSE 0 END)       AS nb_top_annul_1,
    SUM(CASE WHEN CAST(top_annul AS INT) = 0 THEN 1 ELSE 0 END)       AS nb_top_annul_0,
    SUM(CASE WHEN type_porteur = 'Porteur' THEN 1 ELSE 0 END)         AS nb_porteur,
    SUM(CASE WHEN montant_ope IS NULL OR CAST(montant_ope AS DOUBLE) <= 0 THEN 1 ELSE 0 END) AS nb_montant_nul,

    -- type_reseau
    SUM(CASE WHEN LPAD(CAST(type_reseau AS STRING), 2, '0') = '00' THEN 1 ELSE 0 END) AS reseau_00,
    SUM(CASE WHEN LPAD(CAST(type_reseau AS STRING), 2, '0') = '01' THEN 1 ELSE 0 END) AS reseau_01,
    SUM(CASE WHEN LPAD(CAST(type_reseau AS STRING), 2, '0') = '02' THEN 1 ELSE 0 END) AS reseau_02,
    SUM(CASE WHEN LPAD(CAST(type_reseau AS STRING), 2, '0') = '03' THEN 1 ELSE 0 END) AS reseau_03,
    SUM(CASE WHEN LPAD(CAST(type_reseau AS STRING), 2, '0') = '04' THEN 1 ELSE 0 END) AS reseau_04,
    SUM(CASE WHEN LPAD(CAST(type_reseau AS STRING), 2, '0') = '05' THEN 1 ELSE 0 END) AS reseau_05,
    SUM(CASE WHEN LPAD(CAST(type_reseau AS STRING), 2, '0') = '06' THEN 1 ELSE 0 END) AS reseau_06,
    SUM(CASE WHEN LPAD(CAST(type_reseau AS STRING), 2, '0') = '07' THEN 1 ELSE 0 END) AS reseau_07,
    SUM(CASE WHEN LPAD(CAST(type_reseau AS STRING), 2, '0') = '08' THEN 1 ELSE 0 END) AS reseau_08,
    SUM(CASE WHEN LPAD(CAST(type_reseau AS STRING), 2, '0') = '09' THEN 1 ELSE 0 END) AS reseau_09,
    SUM(CASE WHEN LPAD(CAST(type_reseau AS STRING), 2, '0') = '10' THEN 1 ELSE 0 END) AS reseau_10,
    SUM(CASE WHEN LPAD(CAST(type_reseau AS STRING), 2, '0') = '11' THEN 1 ELSE 0 END) AS reseau_11,
    SUM(CASE WHEN LPAD(CAST(type_reseau AS STRING), 2, '0') = '12' THEN 1 ELSE 0 END) AS reseau_12,
    SUM(CASE WHEN type_reseau IS NULL OR LPAD(CAST(type_reseau AS STRING), 2, '0') NOT IN
             ('00','01','02','03','04','05','06','07','08','09','10','11','12') THEN 1 ELSE 0 END) AS reseau_autre,

    -- code_carte_privative
    SUM(CASE WHEN code_carte_privative IS NULL THEN 1 ELSE 0 END) AS priv_null,
    SUM(CASE WHEN code_carte_privative = ''    THEN 1 ELSE 0 END) AS priv_vide,
    SUM(CASE WHEN code_carte_privative = 'AMX' THEN 1 ELSE 0 END) AS priv_amx,
    SUM(CASE WHEN code_carte_privative = 'COS' THEN 1 ELSE 0 END) AS priv_cos,
    SUM(CASE WHEN code_carte_privative = 'CFC' THEN 1 ELSE 0 END) AS priv_cfc,
    SUM(CASE WHEN code_carte_privative = 'CUP' THEN 1 ELSE 0 END) AS priv_cup,
    SUM(CASE WHEN code_carte_privative = 'C24' THEN 1 ELSE 0 END) AS priv_c24,
    SUM(CASE WHEN code_carte_privative = 'DIN' THEN 1 ELSE 0 END) AS priv_din,
    SUM(CASE WHEN code_carte_privative = 'E21' THEN 1 ELSE 0 END) AS priv_e21,
    SUM(CASE WHEN code_carte_privative = 'FCB' THEN 1 ELSE 0 END) AS priv_fcb,
    SUM(CASE WHEN code_carte_privative = 'FCE' THEN 1 ELSE 0 END) AS priv_fce,
    SUM(CASE WHEN code_carte_privative = 'JCB' THEN 1 ELSE 0 END) AS priv_jcb,
    SUM(CASE WHEN code_carte_privative = 'PRA' THEN 1 ELSE 0 END) AS priv_pra,
    SUM(CASE WHEN code_carte_privative = 'PRI' THEN 1 ELSE 0 END) AS priv_pri,
    SUM(CASE WHEN code_carte_privative = '-'   THEN 1 ELSE 0 END) AS priv_tiret,
    SUM(CASE WHEN code_carte_privative NOT IN ('', 'AMX', 'COS', 'CFC', 'CUP', 'C24', 'DIN', 'E21', 'FCB', 'FCE',
             'JCB', 'PRA', 'PRI', '-') THEN 1 ELSE 0 END)     AS priv_autre,

    -- motifs de capture
    SUM(CASE WHEN lib_ope = 'Capture de Carte' AND UPPER(lib_motif_capture_carte) LIKE '%OUBLI%'   THEN 1 ELSE 0 END) AS cap_oubli,
    SUM(CASE WHEN lib_ope = 'Capture de Carte' AND UPPER(lib_motif_capture_carte) LIKE '%LECT%'    THEN 1 ELSE 0 END) AS cap_lecture,
    SUM(CASE WHEN lib_ope = 'Capture de Carte' AND UPPER(lib_motif_capture_carte) LIKE '%CODE%'    THEN 1 ELSE 0 END) AS cap_code,
    SUM(CASE WHEN lib_ope = 'Capture de Carte' AND UPPER(lib_motif_capture_carte) LIKE '%PERDU%'   THEN 1 ELSE 0 END) AS cap_perdue,
    SUM(CASE WHEN lib_ope = 'Capture de Carte' AND UPPER(lib_motif_capture_carte) LIKE '%VOL%'     THEN 1 ELSE 0 END) AS cap_volee,
    SUM(CASE WHEN lib_ope = 'Capture de Carte' AND UPPER(lib_motif_capture_carte) LIKE '%CONSERV%' THEN 1 ELSE 0 END) AS cap_conserver,
    SUM(CASE WHEN lib_ope = 'Capture de Carte' AND (lib_motif_capture_carte IS NULL OR TRIM(lib_motif_capture_carte) = '')
             THEN 1 ELSE 0 END)                                                                                    AS cap_sans_motif,

    -- cassettes : lignes où au moins une position est déclarée à cette valeur (en centimes)
    SUM(CASE WHEN 500   IN (CAST(val_coupurek7_1 AS INT), CAST(val_coupurek7_2 AS INT), CAST(val_coupurek7_3 AS INT), CAST(val_coupurek7_4 AS INT)) THEN 1 ELSE 0 END) AS avec_cassette_5,
    SUM(CASE WHEN 1000  IN (CAST(val_coupurek7_1 AS INT), CAST(val_coupurek7_2 AS INT), CAST(val_coupurek7_3 AS INT), CAST(val_coupurek7_4 AS INT)) THEN 1 ELSE 0 END) AS avec_cassette_10,
    SUM(CASE WHEN 2000  IN (CAST(val_coupurek7_1 AS INT), CAST(val_coupurek7_2 AS INT), CAST(val_coupurek7_3 AS INT), CAST(val_coupurek7_4 AS INT)) THEN 1 ELSE 0 END) AS avec_cassette_20,
    SUM(CASE WHEN 5000  IN (CAST(val_coupurek7_1 AS INT), CAST(val_coupurek7_2 AS INT), CAST(val_coupurek7_3 AS INT), CAST(val_coupurek7_4 AS INT)) THEN 1 ELSE 0 END) AS avec_cassette_50,
    SUM(CASE WHEN 10000 IN (CAST(val_coupurek7_1 AS INT), CAST(val_coupurek7_2 AS INT), CAST(val_coupurek7_3 AS INT), CAST(val_coupurek7_4 AS INT)) THEN 1 ELSE 0 END) AS avec_cassette_100,
    MIN(val_coupurek7_1) AS coupure_min, MAX(val_coupurek7_1) AS coupure_max,
    MIN(nb_billetsk7_1)  AS billets_min, MAX(nb_billetsk7_1)  AS billets_max
FROM `dss_fallback_lbp_dp`.`reporting_t_dlk_mon_nf2_ope_retrait_v1`
