SELECT bloc, cle, valeur, exemple
FROM (
    -- 3 à 6 : types d'opération, réseaux, plans de cassettes, motifs de capture (1re lecture)
    SELECT t.bloc, t.cle, CAST(COUNT(*) AS DOUBLE) AS valeur, CAST(NULL AS STRING) AS exemple
    FROM `dss_fallback_lbp_dp`.`reporting_t_dlk_mon_nf2_ope_retrait_v1` o
    LATERAL VIEW INLINE(ARRAY(
        STRUCT('3_operation', CONCAT_WS(' | ', COALESCE(CAST(o.lib_ope AS STRING), 'vide'),
                                               COALESCE(CAST(o.top_annul AS STRING), 'vide'),
                                               COALESCE(CAST(o.type_porteur AS STRING), 'vide'))),
        STRUCT('4_reseau', CONCAT_WS(' | ', COALESCE(CAST(o.type_reseau AS STRING), 'vide'),
                                            COALESCE(CAST(o.code_carte_privative AS STRING), 'vide'))),
        STRUCT('5_cassettes', CONCAT_WS(' | ', COALESCE(CAST(o.val_coupurek7_1 AS STRING), 'vide'),
                                               COALESCE(CAST(o.val_coupurek7_2 AS STRING), 'vide'),
                                               COALESCE(CAST(o.val_coupurek7_3 AS STRING), 'vide'),
                                               COALESCE(CAST(o.val_coupurek7_4 AS STRING), 'vide'))),
        STRUCT('6_motif_capture', CASE WHEN o.lib_ope = 'Capture de Carte'
                                       THEN COALESCE(CAST(o.lib_motif_capture_carte AS STRING), 'vide') END)
    )) t AS bloc, cle
    WHERE t.cle IS NOT NULL
    GROUP BY t.bloc, t.cle

    UNION ALL

    -- 1 : volumes par année et activité d'un GAB-jour (2e lecture)
    SELECT '1_volumes', CONCAT(a.annee, ' ', m.indicateur), m.valeur, CAST(NULL AS STRING)
    FROM (
        SELECT annee,
               SUM(nb) AS nb_lignes, COUNT(DISTINCT num_automate) AS nb_gab, COUNT(*) AS nb_gab_jours,
               PERCENTILE_APPROX(CAST(nb AS DOUBLE), 0.1) AS p10, PERCENTILE_APPROX(CAST(nb AS DOUBLE), 0.5) AS mediane,
               PERCENTILE_APPROX(CAST(nb AS DOUBLE), 0.9) AS p90, MAX(nb) AS maxi
        FROM (
            SELECT num_automate, jour,
                   CASE WHEN jour LIKE '____-%' THEN SUBSTR(jour, 1, 4) ELSE SUBSTR(jour, 5, 4) END AS annee,
                   COUNT(*) AS nb
            FROM (
                SELECT num_automate,
                       CASE WHEN CAST(date_operation AS STRING) LIKE '____-%'
                            THEN SUBSTR(CAST(date_operation AS STRING), 1, 10)
                            ELSE SUBSTR(LPAD(CAST(date_operation AS STRING), 8, '0'), 1, 8) END AS jour
                FROM `dss_fallback_lbp_dp`.`reporting_t_dlk_mon_nf2_ope_retrait_v1`
            ) j
            GROUP BY num_automate, jour
        ) d
        GROUP BY annee
    ) a
    LATERAL VIEW INLINE(ARRAY(
        STRUCT('nb_lignes', CAST(a.nb_lignes AS DOUBLE)), STRUCT('nb_gab', CAST(a.nb_gab AS DOUBLE)),
        STRUCT('nb_gab_jours', CAST(a.nb_gab_jours AS DOUBLE)),
        STRUCT('ope_par_gab_jour_p10', a.p10), STRUCT('ope_par_gab_jour_mediane', a.mediane),
        STRUCT('ope_par_gab_jour_p90', a.p90), STRUCT('ope_par_gab_jour_max', CAST(a.maxi AS DOUBLE))
    )) m AS indicateur, valeur

    UNION ALL

    -- 2 : formats des colonnes (plus petite et plus grande valeur sur un échantillon de 1 000 lignes)
    SELECT '2_format', f.colonne, CAST(NULL AS DOUBLE), CONCAT(MIN(f.v), '   ...   ', MAX(f.v))
    FROM (SELECT * FROM `dss_fallback_lbp_dp`.`reporting_t_dlk_mon_nf2_ope_retrait_v1` LIMIT 1000) s
    LATERAL VIEW INLINE(ARRAY(
        STRUCT('date_operation', CAST(s.date_operation AS STRING)),
        STRUCT('heure_operation', CAST(s.heure_operation AS STRING)),
        STRUCT('montant_ope', CAST(s.montant_ope AS STRING)),
        STRUCT('encaisse_theorique', CAST(s.encaisse_theorique AS STRING)),
        STRUCT('top_annul', CAST(s.top_annul AS STRING)),
        STRUCT('nb_billetsk7_1', CAST(s.nb_billetsk7_1 AS STRING)),
        STRUCT('val_coupurek7_1', CAST(s.val_coupurek7_1 AS STRING)),
        STRUCT('code_site_implantation', CAST(s.code_site_implantation AS STRING))
    )) f AS colonne, v
    GROUP BY f.colonne

    UNION ALL

    -- 8 : statuts du référentiel
    SELECT '8_statut_referentiel', COALESCE(CAST(statut_gab AS STRING), 'vide'), CAST(COUNT(*) AS DOUBLE), CAST(NULL AS STRING)
    FROM `dss_fallback_lbp_dp`.`referentiel_gab_final`
    GROUP BY statut_gab
) u
ORDER BY bloc, valeur DESC
