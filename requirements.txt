    SELECT 1
    FROM retraits_cos c
    WHERE c.code_bureau = g.code_bureau
      AND c.date_ope    = g.date_ope
      AND c.heure_retrait_seconde BETWEEN g.heure_delivrance_seconde AND g.heure_delivrance_seconde_plus_45
)
