SELECT bloc, valeur, nb
FROM (
    SELECT '0_mois' AS bloc,
           SUBSTR(date_operation, 3, 6) AS valeur,
           COUNT(*) AS nb
    FROM `dss_fallback_lbp_dp`.`reporting_t_dlk_mon_nf2_ope_retrait_v1`
    WHERE SUBSTR(date_operation, 3, 6) IN ('082026', '092026', '102026')
    GROUP BY SUBSTR(date_operation, 3, 6)

    UNION ALL

    SELECT '1_operation',
           CONCAT(COALESCE(lib_ope, 'vide'), ' | annulé=', COALESCE(CAST(top_annul AS STRING), 'vide')),
           COUNT(*)
    FROM `dss_fallback_lbp_dp`.`reporting_t_dlk_mon_nf2_ope_retrait_v1`
    WHERE SUBSTR(date_operation, 3, 6) IN ('082026', '092026', '102026')
    GROUP BY lib_ope, top_annul

    UNION ALL

    SELECT '2_reseau',
           CONCAT('reseau=', COALESCE(CAST(type_reseau AS STRING), 'vide'),
                  ' | privative=', COALESCE(code_carte_privative, 'vide')),
           COUNT(*)
    FROM `dss_fallback_lbp_dp`.`reporting_t_dlk_mon_nf2_ope_retrait_v1`
    WHERE SUBSTR(date_operation, 3, 6) IN ('082026', '092026', '102026')
    GROUP BY type_reseau, code_carte_privative

    UNION ALL

    SELECT '3_cassettes',
           CONCAT(COALESCE(CAST(val_coupurek7_1 AS STRING), 'vide'), ' / ',
                  COALESCE(CAST(val_coupurek7_2 AS STRING), 'vide'), ' / ',
                  COALESCE(CAST(val_coupurek7_3 AS STRING), 'vide'), ' / ',
                  COALESCE(CAST(val_coupurek7_4 AS STRING), 'vide')),
           COUNT(*)
    FROM `dss_fallback_lbp_dp`.`reporting_t_dlk_mon_nf2_ope_retrait_v1`
    WHERE SUBSTR(date_operation, 3, 6) IN ('082026', '092026', '102026')
    GROUP BY val_coupurek7_1, val_coupurek7_2, val_coupurek7_3, val_coupurek7_4

    UNION ALL

    SELECT '4_motif_capture', COALESCE(lib_motif_capture_carte, 'vide'), COUNT(*)
    FROM `dss_fallback_lbp_dp`.`reporting_t_dlk_mon_nf2_ope_retrait_v1`
    WHERE SUBSTR(date_operation, 3, 6) IN ('082026', '092026', '102026')
      AND lib_ope = 'Capture de Carte'
    GROUP BY lib_motif_capture_carte

    UNION ALL

    SELECT '5_heure', SUBSTR(LPAD(CAST(heure_operation AS STRING), 6, '0'), 1, 2), COUNT(*)
    FROM `dss_fallback_lbp_dp`.`reporting_t_dlk_mon_nf2_ope_retrait_v1`
    WHERE SUBSTR(date_operation, 3, 6) IN ('082026', '092026', '102026')
    GROUP BY SUBSTR(LPAD(CAST(heure_operation AS STRING), 6, '0'), 1, 2)
) t
ORDER BY bloc, nb DESC
